### 7.1.3 客户端密钥轮换处理

```java
package top.ilovemyhome.zorasso.core.oidc;

import com.nimbusds.jose.jwk.JWKSet;
import com.nimbusds.jose.jwk.RSAKey;

import java.time.Duration;
import java.time.Instant;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;

/**
 * JWKS 缓存与密钥轮换处理
 */
public class JWKSCacheWithRotation {
    
    private final Map<String, CachedKey> keyCache = new ConcurrentHashMap<>();
    private final Map<String, Instant>> keyExpiration = new ConcurrentHashMap<>();
    
    private final Duration cacheDuration;
    private final Duration gracePeriod;
    
    /**
     * 获取密钥，支持密钥轮换
     * 
     * 策略：
     * 1. 首先尝试获取当前活跃密钥
     * 2. 如果失败，尝试旧密钥（宽限期）
     * 3. 如果都失败，强制刷新 JWKS
     */
    public RSAKey getKey(String jwksUri, String kid) throws Exception {
        // 1. 检查缓存
        CachedKey cached = keyCache.get(kid);
        if (cached != null && !isKeyExpired(cached)) {
            return cached.key;
        }
        
        // 2. 尝试获取所有密钥（可能密钥已轮换）
        try {
            JWKSet jwkSet = fetchJWKS(jwksUri);
            for (var key : jwkSet.getKeys()) {
                if (key instanceof RSAKey rsaKey && 
                    key.getKeyID().equals(kid)) {
                    cacheKey(kid, rsaKey);
                    return rsaKey;
                }
            }
        } catch (Exception e) {
            // 3. 尝试从过期缓存获取（宽限期）
            cached = keyCache.get(kid);
            if (cached != null && isWithinGracePeriod(cached)) {
                return cached.key;
            }
            throw e;
        }
        
        // 4. 密钥不存在
        throw new KeyNotFoundException("Key not found: " + kid);
    }
    
    /**
     * 强制刷新 JWKS
     */
    public void refresh(String jwksUri) throws Exception {
        JWKSet jwkSet = fetchJWKS(jwksUri);
        
        // 清理旧缓存
        keyCache.clear();
        
        // 缓存新密钥
        for (var key : jwkSet.getKeys()) {
            if (key instanceof RSAKey rsaKey) {
                cacheKey(rsaKey.getKeyID(), rsaKey);
            }
        }
    }
    
    private void cacheKey(String kid, RSAKey key) {
        CachedKey cached = new CachedKey(key, Instant.now());
        keyCache.put(kid, cached);
        keyExpiration.put(kid, Instant.now().plus(cacheDuration));
    }
    
    private boolean isKeyExpired(CachedKey cached) {
        Instant expiration = keyExpiration.get(cached.key.getKeyID());
        return expiration != null && Instant.now().isAfter(expiration);
    }
    
    private boolean isWithinGracePeriod(CachedKey cached) {
        Instant cachedTime = cached.cachedAt;
        return Instant.now().isBefore(cachedTime.plus(gracePeriod));
    }
    
    private static class CachedKey {
        final RSAKey key;
        final Instant cachedAt;
        
        CachedKey(RSAKey key, Instant cachedAt) {
            this.key = key;
            this.cachedAt = cachedAt;
        }
    }
}
```

### 7.2.2 Redis 会话存储实现

```java
package top.ilovemyhome.zorasso.core.session;

import com.fasterxml.jackson.databind.ObjectMapper;
import top.ilovemyhome.zorasso.si.security.SessionInfo;

import java.time.Duration;
import java.time.Instant;
import java.util.List;
import java.util.Optional;

/**
 * Redis 会话存储
 */
public class RedisSessionStore implements SessionStore {
    
    private final RedisClient redisClient;
    private final ObjectMapper objectMapper;
    private final Duration sessionIdleTimeout;
    private final Duration sessionAbsoluteTimeout;
    
    private static final String SESSION_KEY_PREFIX = "zora:session:";
    private static final String USER_SESSION_INDEX = "zora:user:sessions:";
    
    /**
     * 创建会话
     */
    @Override
    public SessionInfo createSession(SessionInfo session) {
        String key = SESSION_KEY_PREFIX + session.getId();
        
        try {
            String json = objectMapper.writeValueAsString(session);
            redisClient.setex(key, 
                sessionAbsoluteTimeout.getSeconds(), 
                json);
            
            // 索引用户会话
            String userIndexKey = USER_SESSION_INDEX + session.getUserId();
            redisClient.sadd(userIndexKey, session.getId());
            redisClient.expire(userIndexKey, 
                sessionAbsoluteTimeout.getSeconds());
            
            return session;
        } catch (Exception e) {
            throw new SessionException("Failed to create session", e);
        }
    }
    
    /**
     * 获取会话
     */
    @Override
    public Optional<SessionInfo> getSession(String sessionId) {
        String key = SESSION_KEY_PREFIX + sessionId;
        
        try {
            String json = redisClient.get(key);
            if (json == null) {
                return Optional.empty();
            }
            
            SessionInfo session = objectMapper.readValue(json, SessionInfo.class);
            
            // 检查过期
            if (session.isExpired() || session.isIdleExpired()) {
                deleteSession(sessionId);
                return Optional.empty();
            }
            
            // 更新空闲过期时间
            updateSessionAccessTime(sessionId);
            
            return Optional.of(session);
        } catch (Exception e) {
            throw new SessionException("Failed to get session", e);
        }
    }
    
    /**
     * 更新会话访问时间
     */
    private void updateSessionAccessTime(String sessionId) {
        String key = SESSION_KEY_PREFIX + sessionId;
        
        try {
            String json = redisClient.get(key);
            if (json != null) {
                SessionInfo session = objectMapper.readValue(json, SessionInfo.class);
                session.setLastAccessedAt(Instant.now());
                session.setIdleExpiresAt(Instant.now().plus(sessionIdleTimeout));
                
                redisClient.setex(key, 
                    sessionAbsoluteTimeout.getSeconds(), 
                    objectMapper.writeValueAsString(session));
            }
        } catch (Exception e) {
            // 忽略更新失败
        }
    }
    
    /**
     * 删除会话
     */
    @Override
    public void deleteSession(String sessionId) {
        String key = SESSION_KEY_PREFIX + sessionId;
        
        try {
            // 获取会话以删除用户索引
            String json = redisClient.get(key);
            if (json != null) {
                SessionInfo session = objectMapper.readValue(json, SessionInfo.class);
                
                // 删除用户会话索引
                String userIndexKey = USER_SESSION_INDEX + session.getUserId();
                redisClient.srem(userIndexKey, sessionId);
            }
            
            redisClient.del(key);
        } catch (Exception e) {
            throw new SessionException("Failed to delete session", e);
        }
    }
    
    /**
     * 按用户查询会话
     */
    @Override
    public List<SessionInfo> getSessionsByUser(String userId) {
        String userIndexKey = USER_SESSION_INDEX + userId;
        
        try {
            List<String> sessionIds = redisClient.smembers(userIndexKey);
            
            return sessionIds.stream()
                .map(this::getSession)
                .filter(Optional::isPresent)
                .map(Optional::get)
                .filter(s -> !s.isExpired() && !s.isIdleExpired())
                .toList();
        } catch (Exception e) {
            throw new SessionException("Failed to get user sessions", e);
        }
    }
    
    /**
     * 删除用户所有会话
     */
    @Override
    public void deleteUserSessions(String userId) {
        List<SessionInfo> sessions = getSessionsByUser(userId);
        sessions.forEach(s -> deleteSession(s.getId()));
    }
}
```

### 7.3.1 登出类型

### 7.3.2 RP-Initiated Logout 实现

```java
package top.ilovemyhome.zorasso.core.service;

import top.ilovemyhome.zorasso.si.oidc.OIDCProviderMetadata;

/**
 * 登出服务
 */
public class LogoutService {
    
    private final OIDCProviderMetadata metadata;
    private final String postLogoutRedirectUri;
    private final SessionService sessionService;
    
    /**
     * 构建端点登出 URL
     * 
     * 符合 OpenID Connect RP-Initiated Logout 1.0 规范
     */
    public String buildEndSessionUrl(String idTokenHint, String sessionState) {
        StringBuilder url = new StringBuilder();
        
        if (metadata.getEndSessionEndpoint() == null) {
            throw new IllegalStateException("End session endpoint not configured");
        }
        
        url.append(metadata.getEndSessionEndpoint());
        url.append("?post_logout_redirect_uri=");
        url.append(urlEncode(postLogoutRedirectUri));
        
        if (idTokenHint != null) {
            url.append("&id_token_hint=");
            url.append(urlEncode(idTokenHint));
        }
        
        if (sessionState != null) {
            url.append("&session_state=");
            url.append(urlEncode(sessionState));
        }
        
        return url.toString();
    }
    
    /**
     * 处理登出请求
     */
    public void logout(HttpServletRequest request, HttpServletResponse response) {
        // 1. 获取 Session
        String sessionId = getSessionId(request);
        if (sessionId != null) {
            // 2. 获取会话信息（用于获取 id_token_hint）
            SessionInfo session = sessionService.getSession(sessionId);
            
            if (session != null) {
                // 3. 构建端点登出 URL
                String idTokenHint = session.getIdTokenHint();
                String sessionState = session.getSessionState();
                
                // 4. 清除本地会话
                sessionService.invalidateSession(sessionId);
                
                // 5. 清除 Session Cookie
                clearSessionCookie(response);
                
                // 6. 重定向到 IdP 登出
                if (metadata.getEndSessionEndpoint() != null) {
                    String logoutUrl = buildEndSessionUrl(idTokenHint, sessionState);
                    response.redirect(logoutUrl);
                    return;
                }
            }
        }
        
        // 如果没有 IdP 登出端点，直接返回首页
        response.redirect("/");
    }
    
    /**
     * 处理前端通道登出（Front-Channel Logout）
     */
    public void handleFrontChannelLogout(HttpServletRequest request) {
        // Front-Channel Logout 通过 iframe 加载各应用的登出端点
        // 浏览器负责加载所有 iframe
        String sid = request.getParameter("sid");
        
        if (sid != null) {
            // 根据 SID 查找并清除会话
            sessionService.invalidateSessionBySid(sid);
        }
        
        // 返回空文档（不需要任何内容）
        response.setContentType("text/html");
        response.getWriter().write("<html><body></body></html>");
    }
    
    /**
     * 处理后端通道登出（Back-Channel Logout）
     */
    public void handleBackChannelLogout(String logoutToken) {
        // Back-Channel Logout 是服务器到服务器的请求
```

### 7.5.2 日志脱敏

```java
package top.ilovemyhome.zorasso.core.logging;

import com.fasterxml.jackson.databind.ObjectMapper;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.util.*;

/**
 * 日志脱敏工具
 */
public class LogSanitizer {
    
    private static final Logger logger = LoggerFactory.getLogger("security");
    
    // 需要脱敏的字段
    private static final Set<String> SENSITIVE_FIELDS = Set.of(
        "password",
        "secret",
        "token",
        "access_token",
        "refresh_token",
        "id_token",
        "code",
        "code_verifier",
        "state",
        "nonce",
        "cookie",
        "authorization"
    );
    
    /**
     * 脱敏日志消息
     */
    public static String sanitize(String message) {
        if (message == null) {
            return null;
        }
        
        String result = message;
        
        // 替换敏感字段值
        for (String field : SENSITIVE_FIELDS) {
            // 简单模式：field=value
            result = result.replaceAll(
                "(?i)(" + field + "=)[^&\\s]*", 
                "$1***REDACTED***"
            );
            
            // JSON 模式："field": "value"
            result = result.replaceAll(
                "(?i)(\"" + field + "\"\\s*:\\s*\")[^\"]*(\")", 
                "$1***REDACTED***$2"
            );
        }
        
        return result;
    }
    
    /**
     * 脱敏对象
     */
    public static Map<String, Object> sanitizeObject(Map<String, Object> data) {
        if (data == null) {
            return null;
        }
        
        Map<String, Object> result = new HashMap<>();
        
        for (Map.Entry<String, Object> entry : data.entrySet()) {
            String key = entry.getKey().toLowerCase();
            Object value = entry.getValue();
            
            if (SENSITIVE_FIELDS.contains(key)) {
                result.put(entry.getKey(), "***REDACTED***");
            } else if (value instanceof Map) {
                result.put(entry.getKey(), sanitizeObject((Map<String, Object>) value));
            } else if (value instanceof List) {
                result.put(entry.getKey(), sanitizeList((List<?>) value));
            } else {
                result.put(entry.getKey(), value);
            }
        }
        
        return result;
    }
    
    private static List<?> sanitizeList(List<?> list) {
        return list.stream()
            .map(item -> {
                if (item instanceof Map) {
                    return sanitizeObject((Map<String, Object>) item);
                }
                return item;
            })
            .toList();
    }
    
    /**
     * 安全日志记录
     */
    public static void logSecurityEvent(String event, Map<String, Object> data) {
        Map<String, Object> sanitized = sanitizeObject(data);
        
        // 不记录敏感数据
        logger.info("Security event: {} - {}", event, sanitized);
    }
}
```

### 7.6.2 熔断器实现

```java
package top.ilovemyhome.zorasso.core.circuit;

import java.time.Duration;
import java.time.Instant;
import java.util.concurrent.atomic.AtomicInteger;

/**
 * 熔断器
 */
public class CircuitBreaker {
    
    private final String name;
    private final int failureThreshold;
    private final Duration resetTimeout;
    
    private volatile State state = State.CLOSED;
    private final AtomicInteger failureCount = new AtomicInteger(0);
    private volatile Instant lastFailureTime;
    
    public enum State {
        CLOSED,    // 正常
        OPEN,      // 熔断
        HALF_OPEN  // 半开
    }
    
    public CircuitBreaker(String name, int failureThreshold, Duration resetTimeout) {
        this.name = name;
        this.failureThreshold = failureThreshold;
        this.resetTimeout = resetTimeout;
    }
    
    /**
     * 执行操作
     */
    public <T> T execute(Supplier<T> operation) {
        if (state == State.OPEN) {
            // 检查是否可以进入半开状态
            if (lastFailureTime.plus(resetTimeout).isBefore(Instant.now())) {
                state = State.HALF_OPEN;
            } else {
                throw new CircuitBreakerOpenException(
                    "Circuit breaker is open: " + name);
            }
        }
        
        try {
            T result = operation.get();
            
            // 成功：重置状态
            onSuccess();
            
            return result;
        } catch (Exception e) {
            // 失败：增加失败计数
            onFailure();
            throw e;
        }
    }
    
    private void onSuccess() {
        failureCount.set(0);
        state = State.CLOSED;
    }
    
    private void onFailure() {
        lastFailureTime = Instant.now();
        int count = failureCount.incrementAndGet();
        
        if (count >= failureThreshold) {
            state = State.OPEN;
            // 记录熔断事件
        }
    }
    
    public State getState() {
        return state;
    }
}
```
